#!/bin/bash
# Upload secrets to GCP Secret Manager
# Project: acme-prod
#
# Creates each secret, or adds a new version when it already exists.

upsert_secret() {
  if gcloud secrets describe "$1" --project=acme-prod </dev/null >/dev/null 2>&1; then
    gcloud secrets versions add "$1" --data-file=- --project=acme-prod
  else
    gcloud secrets create "$1" --data-file=- --project=acme-prod
  fi
}

printf '%s' 'demo app' | upsert_secret app-name
printf '%s' 'postgres://admin:s3cr3t@db.internal:5432/app' | upsert_secret database-url
